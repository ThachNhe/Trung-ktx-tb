actor "Người dùng" as User
participant "Frontend"  as FE
participant "Auth API"  as API
database "Database"     as DB

User -> FE    : Nhập email & mật khẩu
FE   -> API   : POST /auth/login {email, password}
API  -> DB    : Truy vấn user theo email
DB   --> API  : Trả về thông tin user
API  -> API   : Kiểm tra mật khẩu (bcrypt)
alt Hợp lệ
  API -> API   : Tạo Access Token + Refresh Token
  API --> FE   : 200 OK + Set-Cookie (tokens)
  FE  --> User : Chuyển hướng tới Dashboard
else Không hợp lệ
  API --> FE   : 401 Unauthorized
  FE  --> User : Hiển thị thông báo lỗi
end
@enduml

' ── [UC3] Activity Diagram – Quy trình đăng ký & duyệt ───────────────────

title [UC3] Activity – Quy trình đăng ký phòng

|Sinh viên|
start
:Xem danh sách phòng trống;
:Chọn phòng và nộp đơn đăng ký;

|Hệ thống|
:Kiểm tra sinh viên đã có đơn đang xử lý?;
if (Đã có đơn pending?) then (Có)
  :Từ chối, trả lỗi 409;
  |Sinh viên|
  :Nhận thông báo lỗi;
  stop
else (Không)
  :Tạo đơn đăng ký (trạng thái: pending);
  :Cập nhật trạng thái phòng -> reserved;
endif

|Nhân viên|
:Xem danh sách đơn đăng ký;
if (Duyệt hay từ chối?) then (Duyệt)
  |Hệ thống|
  :Cập nhật đơn -> approved;
  :Cập nhật phòng -> occupied;
  |Sinh viên|
  :Nhận thông báo phê duyệt;
else (Từ chối)
  |Hệ thống|
  :Cập nhật đơn -> rejected;
  :Cập nhật phòng -> available;
  |Sinh viên|
  :Nhận thông báo từ chối;
endif
stop
@enduml

:Chọn sinh viên & kỳ thanh toán;
